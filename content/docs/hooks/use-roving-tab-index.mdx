## Usage

```tsx
const { getItemProps } = useRovingTabIndex({ orientation: 'horizontal' });
return (
  <div role="toolbar">
    {items.map((item, i) => (
      <button key={item.id} {...getItemProps(i)}>{item.label}</button>
    ))}
  </div>
);
```
