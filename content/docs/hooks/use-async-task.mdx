<HookInstallation name="useAsyncTask" />

## Usage

```tsx
const { run, abort, isPending, data } = useAsyncTask(
  async ({ signal }, id: string) => {
    const res = await fetch(`/api/users/${id}`, { signal });
    return res.json();
  },
);
```
